# Find all player links
# In squad tables format: || [[Player Name]] || or || [[Article|Display Name]] ||

print("Total links:", len(players))

# Extract player names - filter out non-player entries
player_names = []
for wiki_title, display in players:
    name = display.strip() if display.strip() else wiki_title.strip()
    # Skip non-player entries
    skip = ['File:', 'Image:', 'Category:', 'flagicon', 'Flag of', 'Template:', 
            'nat fs', 'kit', 'color', 'Wikipedia:', '#', 'born', 'club']
    if not any(s.lower() in wiki_title.lower() for s in skip):
        if not any(s.lower() in name.lower() for s in skip):
            if len(name) > 3 and name[0].isupper():
                player_names.append((wiki_title, name))

seen = set()
unique_players = []
for wt_title, name in player_names:
    if wt_title not in seen:
        seen.add(wt_title)
        unique_players.append((wt_title, name))

print(f"Unique potential players: {len(unique_players)}")

# Save
with open('/tmp/claude-0/-tmp-cc-football-tp4nfz49/368ec84b-802d-4e15-b0b1-56c85ac0a214/scratchpad/all_wiki_links.txt', 'w') as f:
